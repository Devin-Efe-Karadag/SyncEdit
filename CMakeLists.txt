cmake_minimum_required(VERSION 3.16)
project(SyncEdit VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
option(SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
find_package(Curses REQUIRED)
add_library(core src/rope.cpp src/order_index.cpp src/crdt.cpp src/protocol.cpp src/persistence.cpp src/document.cpp src/util.cpp)
target_include_directories(core PUBLIC include)
add_executable(syncedit src/main.cpp src/peer_manager.cpp src/editor_ui.cpp)
target_include_directories(syncedit PRIVATE ${CURSES_INCLUDE_DIRS})
target_link_libraries(syncedit PRIVATE core ${CURSES_LIBRARIES})
enable_testing()
add_test(NAME cli COMMAND ${CMAKE_COMMAND} -E env python3 ${CMAKE_SOURCE_DIR}/tests/integration/test_cli.py $<TARGET_FILE:syncedit>)
set_tests_properties(cli PROPERTIES TIMEOUT 60)
foreach(t crdt rope protocol persistence storage)
 add_executable(test_${t} tests/test_${t}.cpp)
 target_link_libraries(test_${t} PRIVATE core)
 add_test(NAME ${t} COMMAND test_${t})
 set_tests_properties(${t} PROPERTIES TIMEOUT 120)
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME sessions COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/test_sessions.py $<TARGET_FILE:syncedit>)
set_tests_properties(sessions PROPERTIES TIMEOUT 60)
add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/test_network.py $<TARGET_FILE:syncedit>)
set_tests_properties(integration PROPERTIES TIMEOUT 90)
add_custom_target(sanitizer-test COMMAND ${CMAKE_COMMAND} -S ${CMAKE_SOURCE_DIR} -B ${CMAKE_SOURCE_DIR}/build-sanitize -DSANITIZE=ON -DCMAKE_BUILD_TYPE=Debug COMMAND ${CMAKE_COMMAND} --build ${CMAKE_SOURCE_DIR}/build-sanitize -j2 COMMAND ${CMAKE_CTEST_COMMAND} --test-dir ${CMAKE_SOURCE_DIR}/build-sanitize --output-on-failure)
target_link_libraries(edit_benchmark PRIVATE core)

add_test(NAME randomized_network COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/test_randomized.py $<TARGET_FILE:syncedit>)
set_tests_properties(randomized_network PROPERTIES TIMEOUT 180)
add_executable(recovery_benchmark benchmarks/recovery_benchmark.cpp)
target_link_libraries(recovery_benchmark PRIVATE core)
